class Solution {
1. Squares of a Sorted Array    


 public int[] sortedSquares(int[] nums) {
        int n = nums.length;
        int[] result = new int[n];

        int left = 0;
        int right = n - 1;
        int index = n - 1;

        while (left <= right) {
            if (Math.abs(nums[left]) > Math.abs(nums[right])) {
                result[index] = nums[left] * nums[left];
                left++;
            } else {
                result[index] = nums[right] * nums[right];
                right--;
            }
            index--;
        }

        return result;
    }
}


OUTPUT:


Input
nums =
[-4,-1,0,3,10]
Output
[0,1,9,16,100]
Expected
[0,1,9,16,100]


2. Find Pivot Index 



class Solution {
    public int pivotIndex(int[] nums) {
        int total = 0;
        int left = 0;

        for (int n : nums)
            total += n;

        for (int i = 0; i < nums.length; i++) {
            int right = total - left - nums[i];

            if (left == right)
                return i;

            left += nums[i];
        }

        return -1;
    }
}

nums =
[1,7,3,6,5,6]
