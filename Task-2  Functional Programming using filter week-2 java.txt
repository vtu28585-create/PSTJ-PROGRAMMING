Functional Programming using filter() 
1. Remove Element and 2. Remove Duplicates from Sorted Array


1) class Solution {
    public int removeElement(int[] nums, int val) {
        int k = 0;

        for (int i = 0; i < nums.length; i++) {
            if (nums[i] != val) {
                nums[k] = nums[i];
                k++;
            }
        }

        return k;
    }
}


Input
nums =
[3,2,2,3]
val =
3
Output
[2,2]
Expected
[2,2]



class Solution {
    public int removeDuplicates(int[] nums) {
        if (nums.length == 0) {
            return 0;
        }

        int k = 1;

        for (int i = 1; i < nums.length; i++) {
            if (nums[i] != nums[k - 1]) {
                nums[k] = nums[i];
                k++;
            }
        }

        return k;
    }
}



Input
nums =
[1,1,2]

Output
[1,2]

Expected
[1,2]
