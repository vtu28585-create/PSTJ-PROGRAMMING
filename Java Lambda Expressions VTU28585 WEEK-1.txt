Java Lambda Expressions 

import java.io.*;
import java.util.*;

public class Solution {

    // Functional interface (usually provided in the locked stub)
    interface PerformOperation {
        String getResult(int num);
    }

    // 1. isOdd()
    static PerformOperation isOdd() {
        return (num) -> (num % 2 != 0) ? "ODD" : "EVEN";
    }

    // 2. isPrime()
    static PerformOperation isPrime() {
        return (num) -> {
            if (num < 2) {
                return "COMPOSITE";
            }
            for (int i = 2; i <= Math.sqrt(num); i++) {
                if (num % i == 0) {
                    return "COMPOSITE";
                }
            }
            return "PRIME";
        };
    }

    // 3. isPalindrome()
    static PerformOperation isPalindrome() {
        return (num) -> {
            String original = String.valueOf(num);
            String reversed = new StringBuilder(original).reverse().toString();
            return original.equals(reversed) ? "PALINDROME" : "NOT PALINDROME";
        };
    }

    public static void main(String[] args) {
        Scanner scan = new Scanner(System.in);
        int t = Integer.parseInt(scan.nextLine().trim());

        while (t-- > 0) {
            StringTokenizer st = new StringTokenizer(scan.nextLine());
            int type = Integer.parseInt(st.nextToken());
            int num = Integer.parseInt(st.nextToken());

            PerformOperation operation = null;

            switch (type) {
                case 1:
                    operation = isOdd();
                    break;
                case 2:
                    operation = isPrime();
                    break;
                case 3:
                    operation = isPalindrome();
                    break;
            }

            System.out.println(operation.getResult(num));
        }

        scan.close();
    }
}
