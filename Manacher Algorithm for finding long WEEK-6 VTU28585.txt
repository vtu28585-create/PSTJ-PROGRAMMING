Manacher Algorithm for finding longest palindromic substring 


TASK-1


class Solution {
    public String longestPalindrome(String s) {
        if (s == null || s.length() < 2) {
            return s;
        }

        int start = 0;
        int end = 0;

        for (int i = 0; i < s.length(); i++) {

            // Odd length palindrome
            int len1 = expand(s, i, i);

            // Even length palindrome
            int len2 = expand(s, i, i + 1);

            int len = Math.max(len1, len2);

            if (len > end - start + 1) {
                start = i - (len - 1) / 2;
                end = i + len / 2;
            }
        }

        return s.substring(start, end + 1);
    }

    private int expand(String s, int left, int right) {
        while (left >= 0 && right < s.length()
                && s.charAt(left) == s.charAt(right)) {
            left--;
            right++;
        }

        return right - left - 1;
    }
}

import java.util.*;

public class Solution {

    // Find longest palindromic substring
    static int longestPalindrome(String s) {
        int n = s.length();

        int[] d1 = new int[n]; // Odd length palindromes
        int left = 0, right = -1;

        int max = 0;

        for (int i = 0; i < n; i++) {

            int k = (i > right) ? 1 : Math.min(d1[left + right - i], right - i + 1);

            while (i - k >= 0 && i + k < n &&
                   s.charAt(i - k) == s.charAt(i + k)) {
                k++;
            }

            d1[i] = k--;

            if (i + k > right) {
                left = i - k;
                right = i + k;
            }

            max = Math.max(max, 2 * d1[i] - 1);
        }

        int[] d2 = new int[n]; // Even length palindromes
        left = 0;
        right = -1;

        for (int i = 0; i < n; i++) {

            int k = (i > right) ? 0 : Math.min(d2[left + right - i + 1],
                                                right - i + 1);

            while (i - k - 1 >= 0 && i + k < n &&
                   s.charAt(i - k - 1) == s.charAt(i + k)) {
                k++;
            }

            d2[i] = k--;

            if (i + k > right) {
                left = i - k - 1;
                right = i + k;
            }

            max = Math.max(max, 2 * d2[i]);
        }

        return max;
    }

    public static void main(String[] args) {

        Scanner sc = new Scanner(System.in);

        int n = sc.nextInt();
        String s = sc.next();

        // Double the string so that every rotation can be represented
        String doubled = s + s;

        for (int rotation = 0; rotation < n; rotation++) {

            String current = doubled.substring(rotation, rotation + n);

            System.out.println(longestPalindrome(current));
        }

        sc.close();
    }
}
